from typing import TypedDict, Literal
from langgraph.graph import StateGraph, END
from langchain.chat_models import ChatOpenAI
from langchain.tools import tool
from langchain.agents import create_openai_tools_agent
from langchain.agents.agent_executor import AgentExecutor
from langchain.prompts import ChatPromptTemplate
from langchain.schema import HumanMessage

# -------------------------
# 1. Shared State
# -------------------------

class PortfolioState(TypedDict, total=False):
    user_input: str
    route: str
    research: str
    risk: str
    optimization: str
    final: str


llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)


# -------------------------
# 2. Define Tools
# -------------------------

@tool
def market_news_tool(query: str) -> str:
    """Fetch market news for a given sector."""
    return f"Market news: {query} sector showing high valuation pressure."

@tool
def risk_calculator_tool(tech_weight: float) -> str:
    """Calculate risk level based on tech allocation."""
    if tech_weight > 0.6:
        return "Risk Level: HIGH"
    return "Risk Level: MODERATE"


# -------------------------
# 3. Tool Agents
# -------------------------

def create_tool_agent(system_prompt, tools):
    prompt = ChatPromptTemplate.from_messages([
        ("system", system_prompt),
        ("human", "{input}")
    ])
    agent = create_openai_tools_agent(llm, tools, prompt)
    return AgentExecutor(agent=agent, tools=tools, verbose=False)


research_executor = create_tool_agent(
    "You are a financial research expert. Use tools if helpful.",
    [market_news_tool]
)

risk_executor = create_tool_agent(
    "You are a portfolio risk analyst. Use tools if helpful.",
    [risk_calculator_tool]
)


# -------------------------
# 4. Planner (Dynamic Router)
# -------------------------

def planner(state: PortfolioState):
    prompt = f"""
    Decide which workflow to run:
    - research
    - risk
    - optimize
    - summary

    User request: {state['user_input']}

    Respond with only one word.
    """
    response = llm.invoke([HumanMessage(content=prompt)])
    return {"route": response.content.strip().lower()}


# -------------------------
# 5. Agent Nodes
# -------------------------

def research_node(state: PortfolioState):
    result = research_executor.invoke({"input": state["user_input"]})
    return {"research": result["output"]}


def risk_node(state: PortfolioState):
    result = risk_executor.invoke({"input": state["user_input"]})
    return {"risk": result["output"]}


def optimize_node(state: PortfolioState):
    response = llm.invoke(f"""
    Optimize portfolio based on:
    Research: {state.get('research')}
    Risk: {state.get('risk')}
    """)
    return {"optimization": response.content}


def summary_node(state: PortfolioState):
    return {"final": f"Summary only: {state['user_input']}"}


def aggregator(state: PortfolioState):
    final = f"""
    Research:
    {state.get('research')}

    Risk:
    {state.get('risk')}

    Optimization:
    {state.get('optimization')}
    """
    return {"final": final}


# -------------------------
# 6. Routing Logic
# -------------------------

def route_decision(state: PortfolioState) -> Literal[
    "research", "risk", "optimize", "summary"
]:
    return state["route"]


# -------------------------
# 7. Build Graph
# -------------------------

graph = StateGraph(PortfolioState)

graph.add_node("planner", planner)
graph.add_node("research", research_node)
graph.add_node("risk", risk_node)
graph.add_node("optimize", optimize_node)
graph.add_node("summary", summary_node)
graph.add_node("aggregator", aggregator)

graph.set_entry_point("planner")

graph.add_conditional_edges(
    "planner",
    route_decision,
    {
        "research": "research",
        "risk": "risk",
        "optimize": "research",
        "summary": "summary",
    },
)

# Optimize path = research → risk → optimize → aggregator
graph.add_edge("research", "risk")
graph.add_edge("risk", "optimize")
graph.add_edge("optimize", "aggregator")

# Direct paths
graph.add_edge("research", "aggregator")
graph.add_edge("risk", "aggregator")
graph.add_edge("summary", END)
graph.add_edge("aggregator", END)

app = graph.compile()


# -------------------------
# 8. Run Examples
# -------------------------

print("\n--- Example 1: Risk Only ---")
result = app.invoke({"user_input": "Analyze my portfolio risk"})
print(result["final"])

print("\n--- Example 2: Optimize ---")
result = app.invoke({"user_input": "Optimize my tech-heavy portfolio"})
print(result["final"])
