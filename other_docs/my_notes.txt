 -- List all tables in schema
  SELECT table_name
  FROM information_schema.tables
  WHERE table_schema = 'finance_asst';

  --get column name from table 
  SELECT
      column_name,
      data_type
  FROM information_schema.columns
  WHERE table_schema = 'finance_asst'
  and table_name = 'users'
  ORDER BY table_name, ordinal_position;
